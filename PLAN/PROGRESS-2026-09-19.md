    ```rust
    let has_token = auth_request.access_token.as_ref()
        .is_some_and(|token| !token.is_empty());
    
    // 2. 构建权限上下文
    if has_token {
        // 验证会话 → 查询权限
        match SessionStateApi::verify_login() {
            Ok(session) => {
                match UserPermissionQueryService::get_user_permission_context(uid) {
                    Ok(perm_ctx) => (session, perm_ctx, uid),
                    Err(_) => (session, UserPermissionContext::guest(), uid),
                }
            }
        }
    } else {
        // 游客权限（等级=1）
        (guest_session, UserPermissionContext::guest(), 0)
    }
    
    // 3. 构建 AuthContext 并填充权限信息
    let auth = AuthContext {
        ...
        permission_context: Some(perm_ctx),
    };
    ```

## 🔑 核心特性

### 游客权限流程

```
请求无 access_token
    ↓
网关检查：has_token = false
    ↓
构建游客权限上下文：
  - uid = 0
  - base_level = 1
  - level_name = "GUEST"
  - roles = []
    ↓
创建 AuthContext，permission_context = UserPermissionContext::guest()
    ↓
下层业务检查：auth.has_permission_level(2) → false
    ↓
拒绝写操作，允许读操作
```

### 登录用户权限流程

```
请求带 access_token
    ↓
网关检查：has_token = true
    ↓
verify_login() → SessionContext(uid=123)
    ↓
UserPermissionQueryService::get_user_permission_context(123)
  → 查询 cola_user 权限中心
  → 返回 UserPermissionContext { base_level: 2, ... }
    ↓
创建 AuthContext，permission_context = Some(perm_ctx)
    ↓
下层业务检查：auth.has_permission_level(4) → true if level >= 4
    ↓
根据权限执行相应操作
```
